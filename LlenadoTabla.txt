/* ============================================================
   8. INVENTARIO
   150 REGISTROS
   ============================================================ */

BEGIN

    FOR i IN 1..150 LOOP

        INSERT INTO inventario_stock
        (
            id_inventario,
            id_corte,
            id_proveedor,
            stock_kilos,
            precio_costo_kilo,
            precio_venta_kilo,

            estado_frescura
        )
        VALUES
        (
            'INV' || LPAD(i,7,'0'),

            'COR' || LPAD(MOD(i-1,50)+1,3,'0'),

            'PROV' || LPAD(MOD(i-1,20)+1,3,'0'),

            ROUND(10 + MOD(i*17,150),2),

            ROUND(2500 + MOD(i*731,7000),0),

            ROUND(
                (2500 + MOD(i*731,7000)) * 1.35,
                0
            ),

            SYSDATE - MOD(i,120),

            SYSDATE + MOD(i,30) + 5,

            CASE MOD(i,5)
                WHEN 0 THEN 'Disponible'
                WHEN 1 THEN 'Disponible'
                WHEN 2 THEN 'Fresco'
                WHEN 3 THEN 'Refrigerado'
                ELSE 'Disponible'
            END
        );

    END LOOP;

    COMMIT;

END;

/* ============================================================
   11. ACTUALIZAR EL MONTO TOTAL DE LOS PEDIDOS
   ============================================================ */

UPDATE pedidos p
SET monto_total =
(
    SELECT NVL(
        SUM(
            d.kilos_pedidos *
            d.precio_unitario_aplicado
        ),0
    )
    FROM detalles_pedido d
    WHERE d.id_pedido = p.id_pedido
);

COMMIT;


/* ============================================================
   12. ACTUALIZAR PUNTOS DE FIDELIDAD
   ============================================================ */

UPDATE clientes c
SET puntos_fidelidad =
(
    SELECT NVL(
        FLOOR(SUM(p.monto_total) / 1000),
        0
    )
    FROM pedidos p
    WHERE p.id_cliente = c.id_cliente
);

COMMIT;


/* ============================================================
   13. VERIFICACION DE CANTIDADES
   ============================================================ */

SELECT 'CIUDADES' AS TABLA, COUNT(*) AS CANTIDAD
FROM ciudades
UNION ALL
SELECT 'CATEGORIAS_ANIMAL', COUNT(*)
FROM categorias_animal
UNION ALL
SELECT 'TIPOS_ANIMAL', COUNT(*)
FROM tipos_animal
UNION ALL
SELECT 'EMPLEADOS', COUNT(*)
FROM empleados
UNION ALL
SELECT 'PROVEEDORES', COUNT(*)
FROM proveedores
UNION ALL
SELECT 'CLIENTES', COUNT(*)
FROM clientes
UNION ALL
SELECT 'CORTES', COUNT(*)
FROM cortes
UNION ALL
SELECT 'INVENTARIO_STOCK', COUNT(*)
FROM inventario_stock
UNION ALL
SELECT 'PEDIDOS', COUNT(*)
FROM pedidos
UNION ALL
SELECT 'DETALLES_PEDIDO', COUNT(*)
FROM detalles_pedido
ORDER BY 1;


/* ============================================================
   14. COMPROBAR PEDIDOS CON SUS DETALLES
   ============================================================ */

SELECT
    p.id_pedido,
    c.nombre || ' ' || c.apellido AS cliente,
    e.nombre || ' ' || e.apellido AS empleado,
    ci.nombre AS ciudad_despacho,
    p.monto_total,
    p.fecha_pedido,
    p.estado_pedido,
    p.metodo_pago,
    p.canal_venta
FROM pedidos p
JOIN clientes c
    ON p.id_cliente = c.id_cliente
JOIN empleados e
    ON p.id_empleado = e.id_empleado
JOIN ciudades ci
    ON p.id_ciudad_despacho = ci.id_ciudad
ORDER BY p.fecha_pedido DESC;


/* ============================================================
   15. COMPROBAR INVENTARIO CON CORTES Y PROVEEDORES
   ============================================================ */

SELECT
    i.id_inventario,
    co.nombre AS corte,
    ta.nombre AS tipo_animal,
    ca.letra_categoria AS categoria,
    pr.razon_social AS proveedor,
    i.stock_kilos,
    i.precio_costo_kilo,
    i.precio_venta_kilo,
    i.fecha_ingreso,
    i.fecha_vencimiento,
    i.estado_frescura
FROM inventario_stock i
JOIN cortes co
    ON i.id_corte = co.id_corte
JOIN tipos_animal ta
    ON co.id_tipo_animal = ta.id_tipo_animal
JOIN categorias_animal ca
    ON co.id_categoria_animal = ca.id_categoria_animal
JOIN proveedores pr
    ON i.id_proveedor = pr.id_proveedor
ORDER BY i.id_inventario;
